function imgPlaceholder(img, svg) {
  if (svg) return `<div style="margin-top:1rem;">${svg}</div>`;
  if (!img) return '';
  const lines = img.split('\n');
  const title = lines[0] || '';
  const desc = lines.slice(1).join(' ');
  return `<div class="img-placeholder"><div class="img-title">${title}</div><p class="img-desc">${desc}</p></div>`;
}

function render() {
  deck.innerHTML = '';
  slideData.forEach((s, i) => deck.appendChild(buildSlide(s, i)));
  go(0);
}

function go(n) {
  current = Math.max(0, Math.min(slideData.length - 1, n));
  deck.style.transform = `translateX(-${current * 100}vw)`;
  prevBtn.style.opacity = current === 0 ? '0.3' : '1';
  nextBtn.style.opacity = current === slideData.length - 1 ? '0.3' : '1';
  if (window.MathJax) MathJax.typesetPromise();
}

window.addEventListener('keydown', e => {
  if (e.key === 'ArrowRight' || e.key === ' ') go(current + 1);
  if (e.key === 'ArrowLeft') go(current - 1);
});
prevBtn.addEventListener('click', () => go(current - 1));
